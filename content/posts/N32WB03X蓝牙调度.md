## N32WB03X蓝牙调度

```c
void Blue_Stack_Process(void)
{
    const uint32 CurTime = Tim_GetCurrentTime();

    rwip_schedule();/*蓝牙任务调度*/

    //更新从站和主站的通讯状态
    Refesh_Slave_Condix();
    Refesh_Master_Condix();
    //发送数据超时处理 从机1
    //从站1处于连接状态，持续SEND_TIMEOUT 100ms没有发送数据后会清除从站1的发送缓冲区
    //从站2、主站同理
    if((Blue_Show_State.Slave1_Connect_State == Connect_Connected) && Rdtss_Send_Flg_Get())
    {
        if(Tim_CheckTimeOut(Slave_Send_Time, CurTime,SEND_TIMEOUT))
        {
    		Uart_ClearBlueSendBuff(BLUE_SIMULATE_UART_NUM_SLVAE1);
             Slave_Send_Time = CurTime;
        }
    }
    else{
            Uart_ClearBlueSendBuff(BLUE_SIMULATE_UART_NUM_SLVAE1);
       		Slave_Send_Time = CurTime;
    }
    //...省略从站2、主站类似代码

    //主站通讯超时30s且从站通讯超时30s且复位超时63s，复位蓝牙协议栈
     if(Tim_CheckTimeOut(Blue_Show_State.LastMaster_CommTime, Tim_GetCurrentTime(), BLUE_COMM_TIMEOUT) && \
        Tim_CheckTimeOut(Blue_Show_State.LastSlave_CommTime, Tim_GetCurrentTime(), BLUE_COMM_TIMEOUT) && \
        Tim_CheckTimeOut(Blue_Show_State.ResetTime, Tim_GetCurrentTime(), BLUE_RESET_INV)){
     	 Blue_Show_State.ResetTime = Tim_GetCurrentTime();
     	 Blue_Show_State.MasterReConnTime = Blue_Show_State.ResetTime;
         Blue_Show_State.ResetCnt++;
		Blue_Stacke_Reset();
     }
}
```

## 蓝牙初始化

```c
void app_ble_init(void)
{
    struct ns_stack_cfg_t app_handler = {0};

    app_handler.ble_msg_handler  = app_ble_msg_handler;   //BLE消息处理回调函数
    app_handler.user_msg_handler = app_user_msg_handler;  //用户消息处理回调函数
    app_handler.lsc_cfg          = BLE_LSC_LSI_32000HZ;   //设置低频时钟频率为32kHz
    ns_ble_stack_init(&app_handler);
    ns_ble_radio_power_set(TX_POWER_Pos6_DBM);           //设置蓝牙发射功率为+6dBM

    app_ble_gap_params_init();  //GAP参数初始化
    app_ble_sec_init();         //安全参数初始化
    app_ble_adv_init();         //广播参数初始化
    app_ble_scan_init();        //扫描功能初始化
    app_ble_prf_init();         //协议栈profile初始化

    Blue_Slave_Adv_Start();        //启动从机广播功能
    //Blue_Master_Scan_Start();    //启动主机扫描功能
    Blue_Show_State.Current_State = Slave_State;     //当前状态为从机状态
    Blue_Show_State.Slave1_Coindex = 0xFF;           //初始化从设备1索引号为无效值
    Blue_Show_State.Slave2_Coindex = 0xFF;	  		//初始化从设备2索引号为无效值
}
```

## GAP(通用访问协议)参数初始化

```c
/**
 * @brief  ble GAP initialization
 * @param
 * @return
 * @note
 */
void app_ble_gap_params_init(void)
{
    struct ns_gap_params_t dev_info = {0};
    uint8_t *p_mac = SystemGetMacAddr(); //获取系统的MAC地址
    //get UUID from trim stored
    if(p_mac != NULL)
    {
    	//set the uuid as mac address
        memcpy(dev_info.mac_addr.addr, p_mac , BD_ADDR_LEN);
        memcpy(Blue_Show_State.Self_Mac, p_mac , MAC_LEN_MAX);
        reverse_array(Blue_Show_State.Self_Mac); //反转MAC地址顺序
    }
    else
    {   //如果获取到，使用默认的MAC地址
        memcpy(dev_info.mac_addr.addr, "\x01\x02\x03\x0C\x0D\x06" , BD_ADDR_LEN);
    }

    /* 初始化参数 */
    dev_info.mac_addr_type = GAPM_STATIC_ADDR;       //MAC地址为静态地址
    dev_info.appearance = 0;                         //设备外观类型为0(代表未知类型)
    dev_info.dev_role = GAP_ROLE_ALL;                //支持所有类型，observer、broadcaster、centeral、peripheral
    dev_info.dev_name_len = strlen(CUSTOM_DEVICE_NAME)-1;                      //设置设备名称长度
    memcpy(dev_info.dev_name, CUSTOM_DEVICE_NAME, dev_info.dev_name_len);      //设置设备名称
    memcpy(Blue_Show_State.Self_Name,dev_info.dev_name,dev_info.dev_name_len); //设置设备名称
    dev_info.dev_conn_param.intv_min = MSECS_TO_UNIT(MIN_CONN_INTERVAL,MSECS_UNIT_1_25_MS);  //设置最小连接间隔
    dev_info.dev_conn_param.intv_max = MSECS_TO_UNIT(MAX_CONN_INTERVAL,MSECS_UNIT_1_25_MS);  //设置最大连接间隔
    dev_info.dev_conn_param.latency  = SLAVE_LATENCY;    /*设置从机延迟时间-允许从机在没有数据发送时，跳过一定数目的主机连接事件，这些事件中从机不必回复主机的数据包，以降低功耗*/
	dev_info.dev_conn_param.time_out = MSECS_TO_UNIT(CONN_SUP_TIMEOUT,MSECS_UNIT_10_MS);    //设置连接超时时间
    dev_info.conn_param_update_delay = FIRST_CONN_PARAMS_UPDATE_DELAY;                     //设置更新连接参数延迟时间
	ns_ble_gap_init(&dev_info);
}
```

```C
     ns_sec_init(&sec_init);
```

```c
void app_ble_adv_init(void)
{
    struct ns_adv_params_t user_adv = {0};

    //init advertising data

    user_adv.scan_rsp_data_len = CUSTOM_USER_ADV_SCNRSP_DATA_LEN; //设置扫描响应数据
    memcpy(user_adv.scan_rsp_data,CUSTOM_USER_ADV_SCNRSP_DATA,CUSTOM_USER_ADV_SCNRSP_DATA_LEN);

    user_adv.attach_appearance  = false;            //不附加设备外观信息
    user_adv.attach_name        = true;             //附加设备名称
    user_adv.ex_adv_enable      = false;            //不使用扩展广播
    user_adv.adv_phy            = PHY_1MBPS_VALUE;  //广播物理层：1Mbps

    //init advertising params
    user_adv.directed_adv.enable = false;          //非定向广播

    user_adv.fast_adv.enable    = true;                       //启用快速广播
    user_adv.fast_adv.duration  = CUSTOM_ADV_FAST_DURATION;   //设置广播持续时间
    user_adv.fast_adv.adv_intv  = CUSTOM_ADV_FAST_INTERVAL;   //设置广播间隔

    user_adv.slow_adv.enable    = true;                       //启用慢速广播
    user_adv.slow_adv.duration  = CUSTOM_ADV_SLOW_DURATION;   //设置广播持续时间
    user_adv.slow_adv.adv_intv  = CUSTOM_ADV_SLOW_INTERVAL;   //设置广播间隔

    user_adv.ble_adv_msg_handler = app_ble_adv_msg_handler;   //广播事件处理函数
```

## 扫描初始化

```c
void app_ble_scan_init(void)
{
    struct ns_scan_params_t init = {0};
    static const uint8_t target_name[] =  {0x30, 0x31, 0x12, 0x00, 0xAA, 0x5B};
    init.type               = SCAN_PARAM_TYPE;         //扫描类型
    init.dup_filt_pol       = SCAN_PARAM_DUP_FILT_POL; //重复数据过滤策略
    init.connect_enable     = SCAN_PARAM_CONNECT_EN;   //是否允许连接
    init.prop_active_enable = SCAN_PARAM_PROP_ACTIVE;  //是否主动扫描,可获取扫描响应数据
    init.scan_intv          = SCAN_PARAM_INTV;         //扫描间隔，多久扫一次
    init.scan_wd            = SCAN_PARAM_WD;           //扫描窗口，每次扫描持续多久
    init.duration           = SCAN_PARAM_DURATION;     //总扫描时长
    init.initiating_timeout = 3000;                    //初始化超时时间(3000ms),扫描后建立连接超时时间

    init.filter_type    = SCAN_FILTER_BY_ADDRESS;      //按地址过滤
    init.filter_data    = (uint8_t*)&target_name;      //目标设备MAC地址
    init.ble_scan_data_handler  = app_ble_scan_data_handler;     //扫描到数据时的回调函数
    init.ble_scan_state_handler = app_ble_scan_state_handler;    //扫描到状态变化的回调函数
    ns_ble_scan_init(&init);
}
```

```c
void app_ble_prf_init(void)
{
    //add device informaiton server
    //add battery level server
    //add raw data transmit server(rdts) 添加原始数据传输服务
    ns_ble_add_prf_func_register(app_rdtss_add_rdts);  //注册RDTSS(Raw Data Transmit Server)功能
	ns_ble_add_prf_func_register(app_rdtsc_add_rdts);  //注册RDTSC(Raw Data Transmit Client)功能
}

bool ns_ble_add_prf_func_register(ns_ble_add_prf_func_t func)
{
    if(add_prf_evn.prf_num < BLE_NB_PROFILES) // 检查当前已注册的Profile数量是否小于最大允许的Profile数量
    {
        add_prf_evn.add_prf_func_list[add_prf_evn.prf_num] = func; // 将Profile初始化函数添加到列表中
        add_prf_evn.prf_num++;    // Profile计数器加1
        return true;
    }
    return false;
}
```
